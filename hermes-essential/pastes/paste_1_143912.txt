+        pip3 install $pkg --quiet --user
+    fi

+echo "Starting MoE Router on port 8000..."
+cd "$MOE_DIR"
+nohup python3 -m uvicorn main:app \

+    --port 8000 \

+    echo "✅ MoE Router started (PID: $(cat $PID_FILE))"

+    try:

+        pass

+async def require_auth(api_key: str = Security(API_KEY_HEADER)):
